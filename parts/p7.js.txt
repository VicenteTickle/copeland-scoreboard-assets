    if(sig!==prevSig.s||!prev){prevSig.s=sig;state=next;render(prev)}else{$('sb-stamp').textContent='Last checked '+nowET()+' · polls every '+(POLL_MS/1000)+'s · Score Keeper is the only writer';renderRent()}
